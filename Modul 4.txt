"""
PROGRAM PERPUSTAKAAN MINI
Materi: Modul 1-4 (variabel, pengkondisian, perulangan, function, class & method)

Peta syarat tugas:
+------------------------------+----------------------+-----------------------------+
| Jenis                        | FUNCTION             | METHOD (di class)           |
+------------------------------+----------------------+-----------------------------+
| Return tanpa parameter       | ambil_pilihan()      | hitung_total_stok()         |
| Return berparameter          | cek_status_stok(s)   | cari_buku(judul)            |
| Non-return tanpa parameter   | tampilkan_header()   | tampilkan_daftar()          |
| Non-return berparameter      | garis(panjang)       | tambah_buku(j, p, s)        |
+------------------------------+----------------------+-----------------------------+
Aturan pasangan: function ber-return  -> method pasangannya non-return (dan sebaliknya)
  - cek_status_stok() [return]      <-> tampilkan_status() [non-return]
  - tampilkan_header() [non-return] <-> hitung_total_stok() [return]
"""


# ====================== FUNCTION ======================

# Function NON-RETURN, TANPA parameter
def tampilkan_header():
    print("=" * 40)
    print("      SISTEM PERPUSTAKAAN MINI")
    print("=" * 40)


# Function NON-RETURN, BERPARAMETER
def garis(panjang):
    print("-" * panjang)


# Function RETURN, TANPA parameter
def ambil_pilihan():
    print("\n1. Lihat daftar buku")
    print("2. Tambah buku")
    print("3. Cari buku")
    print("4. Pinjam buku")
    print("5. Kembalikan buku")
    print("6. Lihat total stok")
    print("0. Keluar")
    return input("Pilih menu: ").strip()


# Function RETURN, BERPARAMETER
def cek_status_stok(stok):
    if stok == 0:
        return "HABIS"
    elif stok <= 2:
        return "MENIPIS"
    else:
        return "TERSEDIA"


# ====================== CLASS & METHOD ======================

class Perpustakaan:
    def __init__(self, nama):
        self.nama = nama
        self.buku = []  # list berisi dictionary buku

    # Method NON-RETURN, BERPARAMETER
    def tambah_buku(self, judul, penulis, stok):
        for b in self.buku:
            if b["judul"].lower() == judul.lower():
                b["stok"] += stok
                print(f"Stok '{b['judul']}' ditambah. Stok sekarang: {b['stok']}")
                return
        self.buku.append({"judul": judul, "penulis": penulis, "stok": stok})
        print(f"Buku '{judul}' berhasil ditambahkan.")

    # Method NON-RETURN, TANPA parameter
    def tampilkan_daftar(self):
        if len(self.buku) == 0:
            print("Belum ada buku di perpustakaan.")
            return
        garis(40)
        for i, b in enumerate(self.buku, start=1):
            status = cek_status_stok(b["stok"])  # memanggil function return
            print(f"{i}. {b['judul']} - {b['penulis']}")
            print(f"   Stok: {b['stok']} [{status}]")
        garis(40)

    # Method RETURN, BERPARAMETER
    def cari_buku(self, judul):
        for b in self.buku:
            if judul.lower() in b["judul"].lower():
                return b
        return None

    # Method RETURN, TANPA parameter
    def hitung_total_stok(self):
        total = 0
        for b in self.buku:
            total += b["stok"]
        return total

    # Method NON-RETURN, BERPARAMETER (pasangan dari cek_status_stok)
    def tampilkan_status(self, judul):
        b = self.cari_buku(judul)
        if b is None:
            print("Buku tidak ditemukan.")
        else:
            print(f"Status '{b['judul']}': {cek_status_stok(b['stok'])}")

    # Method NON-RETURN, BERPARAMETER
    def pinjam_buku(self, judul):
        b = self.cari_buku(judul)
        if b is None:
            print("Buku tidak ditemukan.")
        elif b["stok"] == 0:
            print("Maaf, stok buku habis.")
        else:
            b["stok"] -= 1
            print(f"Berhasil meminjam '{b['judul']}'. Sisa stok: {b['stok']}")

    # Method NON-RETURN, BERPARAMETER
    def kembalikan_buku(self, judul):
        b = self.cari_buku(judul)
        if b is None:
            print("Buku tidak ditemukan.")
        else:
            b["stok"] += 1
            print(f"Terima kasih, '{b['judul']}' dikembalikan. Stok: {b['stok']}")


# ====================== PROGRAM UTAMA ======================

def main():
    perpus = Perpustakaan("Perpustakaan Kampus")

    # Data awal (perulangan)
    data_awal = [
        ("Laskar Pelangi", "Andrea Hirata", 3),
        ("Bumi Manusia", "Pramoedya A. Toer", 1),
        ("Negeri 5 Menara", "A. Fuadi", 0),
    ]
    for judul, penulis, stok in data_awal:
        perpus.tambah_buku(judul, penulis, stok)

    tampilkan_header()

    while True:
        pilihan = ambil_pilihan()

        if pilihan == "1":
            perpus.tampilkan_daftar()

        elif pilihan == "2":
            judul = input("Judul   : ")
            penulis = input("Penulis : ")
            try:
                stok = int(input("Stok    : "))
                if stok < 0:
                    print("Stok tidak boleh negatif.")
                else:
                    perpus.tambah_buku(judul, penulis, stok)
            except ValueError:
                print("Stok harus berupa angka!")

        elif pilihan == "3":
            judul = input("Cari judul: ")
            hasil = perpus.cari_buku(judul)
            if hasil:
                print(f"Ditemukan: {hasil['judul']} - {hasil['penulis']} (stok {hasil['stok']})")
                perpus.tampilkan_status(hasil["judul"])
            else:
                print("Buku tidak ditemukan.")

        elif pilihan == "4":
            perpus.pinjam_buku(input("Judul buku yang dipinjam: "))

        elif pilihan == "5":
            perpus.kembalikan_buku(input("Judul buku yang dikembalikan: "))

        elif pilihan == "6":
            print(f"Total seluruh stok buku: {perpus.hitung_total_stok()}")

        elif pilihan == "0":
            print("Terima kasih telah menggunakan program ini!")
            break

        else:
            print("Pilihan tidak valid, coba lagi.")


if __name__ == "__main__":
    main()
